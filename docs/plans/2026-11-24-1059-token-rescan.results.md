## Definitions

- M1: per transcript and window, chars of tool results reading `.superlite/prime.md` or `prime-haiku.md` (Read, or Bash cat/sed/head) plus user messages starting `# superlite prime`; one value per transcript that read it.
- M2: chars of each user text block holding a `# Task`/`# Tasks` heading line with `SUPERLITE=` within 400 chars of it, counted from the heading on (any wrapper before it is not counted).
- M3: per transcript and window, chars of tool results reading `review-packet.md`.
- M4: chars of the tool result of each Bash call holding `run_tests` and `--wait`.
- M5: red results with a log read within the next 5 tool calls, k of n.
- p90 is the nearest-rank value at index round(0.9*(n-1)); mean is truncated to an integer.

```python
#!/usr/bin/env python3
"""Live token re-scan (#1059): aggregate numbers only, nothing from a transcript is printed."""
import glob, json, os, re, statistics, sys
from datetime import datetime, timezone

HOME = os.path.expanduser("~")
ROOTS = [f"{HOME}/.claude/projects"] + sorted(glob.glob(f"{HOME}/.claude-account*/projects"))
MIN_MTIME = datetime(2026, 10, 5, tzinfo=timezone.utc).timestamp()
W_BEFORE = ("2026-10-05T00:00:00Z", "2026-11-15T19:37:20Z")
W_AFTER = ("2026-11-16T00:00:00Z", "9999")
PRIME_RE = re.compile(r"\.superlite/prime(-haiku)?\.md")
HEAD_RE = re.compile(r"^# Tasks? [^\n]*$", re.M)
PACKET_RE = re.compile(r"review-packet\.md")
LOG_RE = re.compile(r"\.log\b")
LOG_CMD = re.compile(r"\b(cat|tail|head|less|grep|sed|awk)\b")
RED_RE = re.compile(r"tests \([^)]*\): \d+ passed, ([1-9]\d*) failed|run_tests crashed|; error: ")
READ_CMD = re.compile(r"\b(cat|sed|head)\b")


def window(ts):
    if not isinstance(ts, str):
        return None
    t = ts.replace(".000Z", "Z")
    t = re.sub(r"\.\d+Z$", "Z", t)
    if W_BEFORE[0] <= t <= W_BEFORE[1]:
        return "before"
    if t >= W_AFTER[0]:
        return "after"
    return None


def text_of(c):
    if isinstance(c, str):
        return c
    if isinstance(c, list):
        return "".join(x.get("text", "") for x in c if isinstance(x, dict))
    return ""


def stats(vals):
    if not vals:
        return dict(n=0, p50=0, p90=0, mean=0, total=0)
    s = sorted(vals)
    p90 = s[min(len(s) - 1, int(round(0.9 * (len(s) - 1))))]
    return dict(n=len(s), p50=int(statistics.median(s)), p90=p90, mean=int(sum(s) / len(s)), total=sum(s))


def main():
    m1, m2, m3, m4 = ({"before": [], "after": []} for _ in range(4))
    m2_shapes = {"before": {}, "after": {}}
    m3_reads = {"before": [], "after": []}
    m3_max = {"before": [], "after": []}
    m1_multi = {"before": 0, "after": 0}
    red = {"before": [0, 0], "after": [0, 0]}  # [reopened, n]
    files = lines = skipped = 0
    for root in ROOTS:
        for path in glob.glob(f"{root}/**/*.jsonl", recursive=True):
            try:
                if os.path.getmtime(path) < MIN_MTIME:
                    continue
            except OSError:
                continue
            files += 1
            calls = {}  # id -> (name, input)
            order = []  # tool_use ids in order
            prime = {}  # window -> [chars, reads]
            packet = {}
            pk = {}  # window -> list of single read sizes
            reds = []  # (window, index in order at that point)
            with open(path, errors="replace") as f:
                for line in f:
                    lines += 1
                    try:
                        d = json.loads(line)
                    except ValueError:
                        skipped += 1
                        continue
                    if not isinstance(d, dict) or not isinstance(d.get("message"), dict):
                        skipped += 1
                        continue
                    w = window(d.get("timestamp"))
                    if w is None:
                        skipped += 1
                        continue
                    content = d["message"].get("content")
                    if isinstance(content, str):
                        content = [{"type": "text", "text": content}]
                    if not isinstance(content, list):
                        continue
                    for b in content:
                        if not isinstance(b, dict):
                            continue
                        t = b.get("type")
                        if t == "tool_use":
                            inp = b.get("input") or {}
                            calls[b.get("id")] = (b.get("name"), inp)
                            order.append(b.get("id"))
                        elif t == "text" and d.get("type") == "user":
                            tx = b.get("text", "")
                            st = tx.lstrip()
                            mt = HEAD_RE.search(tx)
                            while mt and "SUPERLITE=" not in tx[mt.start():mt.start() + 400]:
                                mt = HEAD_RE.search(tx, mt.end())
                            if mt:
                                m2[w].append(len(tx) - mt.start())
                                pre = tx[:mt.start()].rstrip()
                                shape = "start" if not pre else "tag" if pre.splitlines()[-1].lstrip().startswith("<") else "prefix"
                                m2_shapes[w][shape] = m2_shapes[w].get(shape, 0) + 1
                            if st.startswith("# superlite prime"):
                                p = prime.setdefault(w, [0, 0]); p[0] += len(tx); p[1] += 1
                        elif t == "tool_result":
                            name, inp = calls.get(b.get("tool_use_id"), (None, {}))
                            if name is None:
                                continue
                            tx = text_of(b.get("content"))
                            cmd = inp.get("command", "") if name == "Bash" else ""
                            fp = inp.get("file_path", "") if name == "Read" else ""
                            tgt = cmd if name == "Bash" else fp
                            rd = name == "Read" or (name == "Bash" and READ_CMD.search(cmd))
                            if rd and PRIME_RE.search(tgt):
                                p = prime.setdefault(w, [0, 0]); p[0] += len(tx); p[1] += 1
                            if rd and PACKET_RE.search(tgt):
                                p = packet.setdefault(w, [0, 0]); p[0] += len(tx); p[1] += 1
                                pk.setdefault(w, []).append(len(tx))
                            if name == "Bash" and "run_tests" in cmd:
                                if "--wait" in cmd:
                                    m4[w].append(len(tx))
                                if RED_RE.search(tx):
                                    reds.append((w, len(order)))
            for w, (chars, reads) in prime.items():
                m1[w].append(chars)
                if reads >= 2:
                    m1_multi[w] += 1
            for w, (chars, reads) in packet.items():
                m3[w].append(chars)
                m3_reads[w].append(reads)
                m3_max[w].append(max(pk[w]))
            for w, idx in reds:
                red[w][1] += 1
                hit = False
                for cid in order[idx:idx + 5]:
                    name, inp = calls[cid]
                    if name == "Read" and LOG_RE.search(inp.get("file_path", "")):
                        hit = True
                    elif name == "Bash":
                        c = inp.get("command", "")
                        if LOG_CMD.search(c) and LOG_RE.search(c):
                            hit = True
                red[w][0] += hit
    out = {"files": files, "lines": lines, "skipped": skipped, "m1_multi": m1_multi, "red": red, "m2_shapes": m2_shapes,
           "m3_reads_p50": {w: stats(v)["p50"] for w, v in m3_reads.items()},
           "m3_max_p50": {w: stats(v)["p50"] for w, v in m3_max.items()}}
    for k, m in (("M1", m1), ("M2", m2), ("M3", m3), ("M4", m4)):
        out[k] = {w: stats(v) for w, v in m.items()}
    print(json.dumps(out, indent=1))


if __name__ == "__main__":
    main()
```
